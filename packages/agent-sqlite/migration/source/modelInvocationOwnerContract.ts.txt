export type ModelInvocationOwnerSnapshot = {
  id: string;
  tenantId: string;
  actorId: string;
  status: string;
  leaseOwner: string | null;
  leaseUntil: Date | null;
  fencingToken: number;
};

export const MODEL_INVOCATION_OWNER_PREFLIGHT_ERROR_CODES = [
  'MODEL_INVOCATION_OWNER_NOT_FOUND',
  'MODEL_INVOCATION_OWNER_TENANT_MISMATCH',
  'MODEL_INVOCATION_OWNER_ACTOR_MISMATCH',
  'MODEL_INVOCATION_OWNER_STATE_INVALID',
  'MODEL_INVOCATION_OWNER_FENCED',
  'MODEL_INVOCATION_OWNER_LEASE_EXPIRED',
  'MODEL_INVOCATION_OWNER_RECOVERY_UNREGISTERED',
] as const;

export function isModelInvocationOwnerPreflightErrorCode(value: unknown) {
  return (MODEL_INVOCATION_OWNER_PREFLIGHT_ERROR_CODES as readonly unknown[]).includes(value);
}

export class ModelInvocationOwnerValidationError extends Error {
  readonly code: string;

  constructor(code: string, message: string) {
    super(`${code}: ${message}`);
    this.name = 'ModelInvocationOwnerValidationError';
    this.code = code;
  }
}

export function evaluateModelInvocationOwnerCallBudget(params: {
  hasExactReplay: boolean;
  used: number;
  limit: number;
}) {
  if (params.hasExactReplay) return 'ALLOW_EXACT_REPLAY' as const;
  return params.used >= params.limit ? 'REJECT_NEW_INVOCATION' as const : 'ALLOW_NEW_INVOCATION' as const;
}

export function validateModelInvocationOwnerSnapshot(params: {
  owner: ModelInvocationOwnerRef;
  tenantId: string;
  actorId: string;
  snapshot: ModelInvocationOwnerSnapshot | null;
  now?: Date;
}) {
  const spec = MODEL_INVOCATION_OWNER_REGISTRY[params.owner.kind];
  const snapshot = params.snapshot;
  if (!snapshot || snapshot.id !== params.owner.id) {
    throw new ModelInvocationOwnerValidationError(
      'MODEL_INVOCATION_OWNER_NOT_FOUND',
      '模型调用对应的持久领域任务不存在',
    );
  }
  if (snapshot.tenantId !== params.tenantId) {
    throw new ModelInvocationOwnerValidationError(
      'MODEL_INVOCATION_OWNER_TENANT_MISMATCH',
      '模型调用与领域任务不属于同一租户',
    );
  }
  if (snapshot.actorId !== params.actorId) {
    throw new ModelInvocationOwnerValidationError(
      'MODEL_INVOCATION_OWNER_ACTOR_MISMATCH',
      '模型调用与领域任务不属于同一账号',
    );
  }
  if (!(spec.allowedStatuses as readonly string[]).includes(snapshot.status)) {
    throw new ModelInvocationOwnerValidationError(
      'MODEL_INVOCATION_OWNER_STATE_INVALID',
      `领域任务状态 ${snapshot.status} 不允许发起模型调用`,
    );
  }
  if (
    spec.requiresLease
    && (
      snapshot.leaseOwner !== params.owner.expectedLeaseOwner
      || snapshot.fencingToken !== params.owner.expectedFencingToken
    )
  ) {
    throw new ModelInvocationOwnerValidationError(
      'MODEL_INVOCATION_OWNER_FENCED',
      '领域任务的当前 Worker 或 fencing token 与调用方不一致',
    );
  }
  if (
    spec.requiresLease
    && (!snapshot.leaseUntil || snapshot.leaseUntil <= (params.now || new Date()))
  ) {
    throw new ModelInvocationOwnerValidationError(
      'MODEL_INVOCATION_OWNER_LEASE_EXPIRED',
      '领域任务租约已经过期，禁止向模型服务商发送请求',
    );
  }
  if (!spec.exactRecovery.trim()) {
    throw new ModelInvocationOwnerValidationError(
      'MODEL_INVOCATION_OWNER_RECOVERY_UNREGISTERED',
      '领域任务没有登记精确恢复路径',
    );
  }
  return spec;
}
